** header-line
#+begin_src emacs-lisp
  (defun my/insert-header-line (text &optional width align fill)
    "Insert a header line containing TEXT padded to WIDTH with FILL char.
  ALIGN is one of `left', `right' or `center' (default `left').
  WIDTH defaults to 80 and FILL defaults to ?─.

  The line is inserted at point and runs *up to* column WIDTH, so any
  text already before the cursor (e.g. a `# ' comment opener) counts
  toward the total — the result still ends at column WIDTH, not beyond.

  Called interactively this is the quick version: it prompts only for
  TEXT and uses the defaults for the rest.  See
  `my/insert-header-line-verbose' to be prompted for everything.

  Example: TEXT \"BOOTSTRAP\" yields
    ─── BOOTSTRAP ───────────────────────────────────────────────────────────────"
    (interactive (list (read-string "Header text: ")))
    (let* ((width (max 0 (- (or width 80) (current-column))))
           (align (or align 'left))
           (fill  (or fill ?─))
           (label (if (string-empty-p text) "" (concat " " text " ")))
           (pad   (max 0 (- width (length label))))
           (lead  (min 3 pad))
           (line
            (pcase align
              ('right  (concat (make-string (- pad lead) fill)
                               label
                               (make-string lead fill)))
              ('center (let ((l (/ pad 2)))
                         (concat (make-string l fill)
                                 label
                                 (make-string (- pad l) fill))))
              (_       (concat (make-string lead fill)
                               label
                               (make-string (- pad lead) fill))))))
      (insert line)))

  (defun my/insert-header-line-verbose (text width align fill)
    "Like `my/insert-header-line' but prompt for every argument.
  Empty answers fall back to the defaults (width 80, left, ?─)."
    (interactive
     (list (read-string "Header text: ")
           (read-number "Width: " 80)
           (intern (completing-read "Align: " '("left" "right" "center")
                                    nil t nil nil "left"))
           (let ((s (read-string "Fill char (default ─): ")))
             (if (string-empty-p s) ?─ (aref s 0)))))
    (my/insert-header-line text width align fill))

  (global-set-key (kbd "C-c -") #'my/insert-header-line)
  (global-set-key (kbd "C-c _") #'my/insert-header-line-verbose)
#+end_src
** box
#+begin_src emacs-lisp
  (defvar my/box-styles
    '((rounded ?─ ?│ ?╭ ?╮ ?╰ ?╯)
      (square  ?─ ?│ ?┌ ?┐ ?└ ?┘)
      (double  ?═ ?║ ?╔ ?╗ ?╚ ?╝)
      (heavy   ?━ ?┃ ?┏ ?┓ ?┗ ?┛)
      (ascii   ?= ?| ?+ ?+ ?+ ?+))
    "Character sets for `my/insert-box', each (HORIZ VERT TL TR BL BR).")

  (defun my/insert-box (text &optional width align style)
    "Insert a three-line box around TEXT using box-drawing characters.
  STYLE is a key in `my/box-styles' (default `rounded'): rounded, square,
  double, heavy or ascii.  ALIGN positions TEXT within the box: `left'
  (default), `right' or `center'.  WIDTH defaults to 80.

  The box runs *up to* column WIDTH.  Whatever precedes the cursor on the
  current line — indentation, a `# ' / `;; ' comment opener, etc. — is
  carried onto the lower two lines, so the box stays commented and
  aligned.

  Called interactively this is the quick version: it prompts only for
  TEXT and uses the defaults for the rest.  See `my/insert-box-verbose'
  to be prompted for everything.

  Example: TEXT \"BOOTSTRAP\" yields
    ╭──────────────────────────────────────────────────────────────────────────╮
    │ BOOTSTRAP                                                                  │
    ╰──────────────────────────────────────────────────────────────────────────╯"
    (interactive (list (read-string "Box text: ")))
    (let* ((prefix (buffer-substring-no-properties (line-beginning-position) (point)))
           (col    (current-column))
           (width  (max 4 (- (or width 80) col)))
           (align  (or align 'left))
           (chars  (cdr (assq (or style 'rounded) my/box-styles)))
           (horiz  (nth 0 chars))
           (vert   (nth 1 chars))
           (tl     (nth 2 chars))
           (tr     (nth 3 chars))
           (bl     (nth 4 chars))
           (br     (nth 5 chars))
           (inner  (- width 2))
           (top    (concat (string tl) (make-string inner horiz) (string tr)))
           (bottom (concat (string bl) (make-string inner horiz) (string br)))
           (avail  (max 0 (- inner 2)))
           (txt    (truncate-string-to-width text avail))
           (gap    (max 0 (- avail (string-width txt))))
           (content
            (pcase align
              ('right  (concat (make-string gap ?\s) txt))
              ('center (let ((l (/ gap 2)))
                         (concat (make-string l ?\s) txt (make-string (- gap l) ?\s))))
              (_       (concat txt (make-string gap ?\s)))))
           (mid (concat (string vert) " " content " " (string vert))))
      (insert top "\n" prefix mid "\n" prefix bottom)))

  (defun my/insert-box-verbose (text width align style)
    "Like `my/insert-box' but prompt for width, alignment and style too.
  Empty answers fall back to the defaults (width 80, left, rounded)."
    (interactive
     (list (read-string "Box text: ")
           (read-number "Width: " 80)
           (intern (completing-read "Align: " '("left" "right" "center")
                                    nil t nil nil "left"))
           (intern (completing-read
                    "Style: " (mapcar (lambda (s) (symbol-name (car s))) my/box-styles)
                    nil t nil nil "rounded"))))
    (my/insert-box text width align style))

  (global-set-key (kbd "C-c =") #'my/insert-box)
  (global-set-key (kbd "C-c +") #'my/insert-box-verbose)
#+end_src

  (defun ruff-format-buffer ()
    (interactive)
    (let* ((filename (or (buffer-file-name) "buffer.py"))
           (out-buf (generate-new-buffer " *ruff-output*"))
           (exit-code (call-process-region
                       (point-min) (point-max)
                       "ruff" nil out-buf nil
                       "format" "--stdin-filename" filename "-")))
      (if (zerop exit-code)
          (let ((new-content (with-current-buffer out-buf (buffer-string))))
            (if (string= new-content (buffer-substring-no-properties (point-min) (point-max)))
                (message "All checks passed")
              (let ((pos (point)))
                (erase-buffer)
                (insert new-content)
                (goto-char pos))))
        (message "ruff format: failed with exit code %d" exit-code))
      (kill-buffer out-buf)))

  (defun ruff-fix-buffer ()
    (interactive)
    (let* ((original (buffer-substring-no-properties (point-min) (point-max)))
           (tmp-file (make-temp-file "ruff-fix-" nil ".py")))
      (write-region (point-min) (point-max) tmp-file nil 'silent)
      (let ((exit-code (call-process "ruff" nil nil nil
                                     "check" "--fix" tmp-file)))
        (if (zerop exit-code)
            (let ((new-content (with-temp-buffer
                                 (insert-file-contents tmp-file)
                                 (buffer-string))))
              (if (string= new-content original)
                  (message "All checks passed")
                (let ((pos (point)))
                  (erase-buffer)
                  (insert new-content)
                  (goto-char pos))))
          (message "ruff fix: failed with exit code %d" exit-code)))
      (delete-file tmp-file)))

  (defun ruff-on-save ()
    (ruff-fix-buffer)
    (ruff-format-buffer))

  (defun ruff-enable-on-save ()

** format on save
   Project-aware: walks up from the buffer to find a formatter
   config and invokes the matching tool over stdin. No-ops silently
   when no config is found, so opening a JS/TS file in a project
   that doesn't ship a formatter doesn't surprise you.

#+begin_src emacs-lisp
  (defvar js-ts-biome-configs '("biome.json" "biome.jsonc"))

  (defvar js-ts-prettier-configs
    '(".prettierrc" ".prettierrc.json" ".prettierrc.yaml" ".prettierrc.yml"
      ".prettierrc.json5" ".prettierrc.js" ".prettierrc.cjs"
      ".prettierrc.mjs" ".prettierrc.toml"
      "prettier.config.js" "prettier.config.cjs" "prettier.config.mjs"))

  (defvar js-ts-dprint-configs '("dprint.json" ".dprint.json"))

  (defun js-ts--has-config-p (dir filenames)
    (seq-some (lambda (f) (file-exists-p (expand-file-name f dir))) filenames))

  (defun js-ts--find-config-root (filenames)
    (let ((start (or (and buffer-file-name (file-name-directory buffer-file-name))
                     default-directory)))
      (locate-dominating-file
       start (lambda (d) (js-ts--has-config-p d filenames)))))

  (defun js-ts--resolve-bin (root tool)
    "Prefer ROOT/node_modules/.bin/TOOL when present, else TOOL on $PATH."
    (let ((local (expand-file-name (concat "node_modules/.bin/" tool) root)))
      (if (file-executable-p local) local tool)))

  (defun js-ts--detect-formatter ()
    "Return (CMD ARGS...) for the project's formatter, or nil."
    (let ((file (or (buffer-file-name) "buffer.tsx")))
      (cond
       ((when-let ((root (js-ts--find-config-root js-ts-biome-configs)))
          `(,(js-ts--resolve-bin root "biome")
            "format" "--stdin-file-path" ,file)))
       ((when-let ((root (js-ts--find-config-root js-ts-prettier-configs)))
          `(,(js-ts--resolve-bin root "prettier")
            "--stdin-filepath" ,file)))
       ((when-let ((root (js-ts--find-config-root js-ts-dprint-configs)))
          `(,(js-ts--resolve-bin root "dprint")
            "fmt" "--stdin" ,file))))))

  (defun js-ts-format-buffer ()
    "Format the current buffer with the project's configured formatter."
    (interactive)
    (if-let ((cmd (js-ts--detect-formatter)))
        (let* ((out-buf (generate-new-buffer " *js-ts-format*"))
               (err-file (make-temp-file "js-ts-format-err-"))
               (exit-code (apply #'call-process-region
                                 (point-min) (point-max)
                                 (car cmd) nil (list out-buf err-file) nil
                                 (cdr cmd))))
          (unwind-protect
              (if (zerop exit-code)
                  (let ((new-content (with-current-buffer out-buf (buffer-string)))
                        (old-content (buffer-substring-no-properties
                                      (point-min) (point-max))))
                    (unless (string= new-content old-content)
                      (let ((pos (point)))
                        (erase-buffer)
                        (insert new-content)
                        (goto-char (min pos (point-max))))))
                (message "js-ts format: %s failed (exit %d): %s"
                         (car cmd) exit-code
                         (with-temp-buffer
                           (insert-file-contents err-file)
                           (string-trim (buffer-string)))))
            (kill-buffer out-buf)
            (delete-file err-file)))
      (when (called-interactively-p 'any)
        (message "js-ts: no project formatter config found"))))

  (defun js-ts-enable-format-on-save ()
    (add-hook 'before-save-hook #'js-ts-format-buffer nil t))

  (dolist (h '(js-ts-mode-hook
               typescript-ts-mode-hook
               tsx-ts-mode-hook))
    (add-hook h #'js-ts-enable-format-on-save))

    :straight (:host github :repo "leon-barrett/just-ts-mode.el")

  (defun my/just-format-buffer ()
    "Format the current justfile buffer in place with `just --fmt'."
    (interactive)
    (when (executable-find "just")
      (let ((tmp (make-temp-file "just-fmt-" nil ".just"))
            (err-file (make-temp-file "just-fmt-err-")))
        (unwind-protect
            (progn
              (write-region (point-min) (point-max) tmp nil 'silent)
              (let ((exit (call-process "just" nil (list nil err-file) nil

                                 (insert-file-contents tmp)

  All jinja files -- standalone (=*.jinja=, =*.jinja2=, =*.j2=) and
  host-templated (=*.py.j2=, =*.tsx.j2=, ...) -- use plain =jinja2-mode=.
  We dropped the previous polymode host+inner setup: it was brittle, had
  to suppress eglot/flymake, and never parsed cleanly because a templated

* jinja host highlighting (tree-sitter injection)
  Host-templated files (=*.py.j2=, =*.tsx.j2=, ...) use =jinja-ts-mode=,
  which highlights the *host* language with its real tree-sitter parser
  while leaving the jinja tags alone. There is no jinja tree-sitter
  grammar involved: we run the host language's parser over just the spans
  *outside* the jinja tags (computed by regexp), reusing the host
  =*-ts-mode='s font-lock, and overlay the tags via jit-lock. The embed
  language is chosen from the inner extension. Plain =*.j2= / =*.jinja=
  (no inner extension) stay in =jinja2-mode=.
#+begin_src emacs-lisp
  (defvar jinja-ts-embed-languages
    '(("py"   . (python     . python-ts-mode))
      ("pyi"  . (python     . python-ts-mode))
      ("ts"   . (typescript . typescript-ts-mode))
      ("tsx"  . (tsx        . tsx-ts-mode))
      ("jsx"  . (tsx        . tsx-ts-mode))
      ("js"   . (javascript . js-ts-mode))
      ("css"  . (css        . css-ts-mode))
      ("json" . (json       . json-ts-mode))
      ("yaml" . (yaml       . yaml-ts-mode))
      ("yml"  . (yaml       . yaml-ts-mode)))
    "Inner extension -> (TS-LANGUAGE . TS-MODE) for jinja host injection.")

  (defconst jinja-ts--tag-re "{%[-+]?.*?[-+]?%}\\|{{.*?}}\\|{#.*?#}"
    "Regexp matching a single jinja tag (statement / expression / comment).")

  (defvar-local jinja-ts--embed-parser nil)

  (defun jinja-ts--entry ()
    "Return (LANG . MODE) for the current buffer's inner extension, or nil."
    (when buffer-file-name
      (let ((name (file-name-nondirectory buffer-file-name)))
        (when (string-match
               "\\.\\([^.]+\\)\\.\\(?:j2\\|jinja2?\\|jinja\\)\\'" name)
          (cdr (assoc (match-string 1 name) jinja-ts-embed-languages))))))

  (defun jinja-ts--host-ranges ()
    "Buffer ranges (BEG . END) that lie outside jinja tags."
    (save-excursion
      (goto-char (point-min))
      (let ((pos (point-min)) ranges)
        (while (re-search-forward jinja-ts--tag-re nil t)
          (when (> (match-beginning 0) pos)
            (push (cons pos (match-beginning 0)) ranges))
          (setq pos (match-end 0)))
        (when (< pos (point-max))
          (push (cons pos (point-max)) ranges))
        (or (nreverse ranges) (list (cons (point-min) (point-max)))))))

  (defun jinja-ts--update-ranges (&rest _)
    (when (treesit-parser-p jinja-ts--embed-parser)
      (treesit-parser-set-included-ranges
       jinja-ts--embed-parser (jinja-ts--host-ranges))))

  (defvar jinja-ts--config-cache (make-hash-table :test 'eq)
    "Cache of host language -> (FONT-LOCK-SETTINGS . FEATURE-LIST).")

  (defun jinja-ts--host-config (lang mode-fn)
    (or (gethash lang jinja-ts--config-cache)
        (puthash lang
                 (with-temp-buffer
                   (funcall mode-fn)
                   (cons treesit-font-lock-settings
                         treesit-font-lock-feature-list))
                 jinja-ts--config-cache)))

  (defun jinja-ts--fontify-tags (start end)
    "Overlay jinja tag faces; runs as a jit-lock function after treesit."
    (save-excursion
      (goto-char start)
      (while (re-search-forward jinja-ts--tag-re end t)
        (let ((face (cond ((eq (char-after (1+ (match-beginning 0))) ?#)
                           'font-lock-comment-face)
                          (t 'font-lock-preprocessor-face))))
          (put-text-property (match-beginning 0) (match-end 0) 'face face)))))

  (define-derived-mode jinja-ts-mode prog-mode "Jinja[ts]"
    "Highlight a Jinja template's host language via tree-sitter injection."
    (let* ((entry (jinja-ts--entry))
           (lang (car entry))
           (mode-fn (cdr entry)))
      (when (and lang (fboundp mode-fn) (treesit-ready-p lang))
        (let ((cfg (jinja-ts--host-config lang mode-fn))
              (parser (treesit-parser-create lang)))
          (setq jinja-ts--embed-parser parser)
          (treesit-parser-set-included-ranges parser (jinja-ts--host-ranges))
          (setq-local treesit-primary-parser parser
                      treesit-font-lock-settings (car cfg)
                      treesit-font-lock-feature-list (cdr cfg)
                      treesit-font-lock-level 4)
          (treesit-major-mode-setup)
          ;; treesit-major-mode-setup may reset ranges; reapply.
          (treesit-parser-set-included-ranges parser (jinja-ts--host-ranges))
          (add-hook 'after-change-functions #'jinja-ts--update-ranges nil t)))
      ;; Tag highlighting runs after treesit's fontification.
      (jit-lock-register #'jinja-ts--fontify-tags)
      (add-hook 'jit-lock-functions #'jinja-ts--fontify-tags t t)))

  ;; Route host-templated files to jinja-ts-mode (front of auto-mode-alist
  ;; so these win over jinja2-mode's plain "\\.j2\\'").
  (dolist (ext (mapcar #'car jinja-ts-embed-languages))
    (add-to-list 'auto-mode-alist
                 (cons (format "\\.%s\\.\\(?:j2\\|jinja2?\\|jinja\\)\\'"
                               (regexp-quote ext))
                       'jinja-ts-mode)))
#+end_src

** todo keywords
   Richer than the default =TODO|DONE=. =BLOCKED= keeps started-but-
   stuck items visually distinct from actively-progressing ones. Org
   forbids spaces in keyword names, so =IN-PROGRESS= uses a hyphen.
   The bracketed letters are =org-todo-keyword= fast-access shortcuts:
   =C-c C-t <letter>= jumps straight to that state.

   #+begin_src emacs-lisp
     (setq org-todo-keywords
           '((sequence "TODO(t)" "IN-PROGRESS(i)" "BLOCKED(b)"
                       "|" "DONE(d)")))

     (setq org-todo-keyword-faces
           '(("TODO"        . (:inherit org-todo))
             ("IN-PROGRESS" . (:inherit org-todo :foreground "#f0c674"))
             ("BLOCKED"     . (:inherit org-todo :foreground "#cc6666"))
             ("DONE"        . (:inherit org-done))))
   #+end_src

** agenda prefix
   Replace the default =%c= (org category, which for us is just the
   filename base and collapses work/inbox and personal/inbox to the
   same string) with a path-derived label: =[work]=, =[personal]=,
   =[proj:<slug>]=, etc. See =my/org-agenda-entry-label=.

   =org-agenda-todo-keyword-format= pads the TODO state to a fixed
   width so headlines line up regardless of which keyword they're in.
   =11= matches the longest keyword (=IN-PROGRESS=); bump it if you
   add a longer one.

   #+begin_src emacs-lisp
     (require 'my-org-agenda)
     (setq org-agenda-prefix-format
           '((agenda . " %i %-14(my/org-agenda-entry-label)%?-12t% s")
             (todo   . " %i %-14(my/org-agenda-entry-label)")
             (tags   . " %i %-14(my/org-agenda-entry-label)")
             (search . " %i %-14(my/org-agenda-entry-label)")))
     (setq org-agenda-todo-keyword-format "%-11s")
   #+end_src
